// チャートの初期化 function initChart() { const chartElement = document.getElementById('chart'); chart = LightweightCharts.createChart(chartElement, { layout: { background: { type: 'solid', color: '#1e1e24' }, textColor: '#d1d5db', }, grid: { vertLines: { color: '#2a2a32' }, horzLines: { color: '#2a2a32' }, }, crosshair: { mode: LightweightCharts.CrosshairMode.Normal, }, // 🇯🇵 日本時間（JST）表示の指定 localization: { locale: 'ja-JP', timeFormatter: time => { const date = new Date(time * 1000); const hours = date.getHours().toString().padStart(2, '0'); const minutes = date.getMinutes().toString().padStart(2, '0'); return `${hours}:${minutes}`; }, }, timeScale: { timeVisible: true, secondsVisible: false, // X軸目盛りの時刻フォーマット tickMarkFormatter: (time, tickType, locale) => { const date = new Date(time * 1000); const hours = date.getHours().toString().padStart(2, '0'); const minutes = date.getMinutes().toString().padStart(2, '0'); return `${hours}:${minutes}`; } }, }); candleSeries = chart.addCandlestickSeries({ upColor: '#26a69a', downColor: '#ef5350', borderVisible: false, wickUpColor: '#26a69a', wickDownColor: '#ef5350', }); vwapSeries = chart.addLineSeries({ color: '#ffb300', lineWidth: 2, title: 'VWAP', }); window.addEventListener('resize', () => { if (chart && chartElement) { chart.applyOptions({ width: chartElement.clientWidth }); } }); }
